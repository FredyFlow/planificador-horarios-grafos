\section{Planteamiento del Problema, Objetivos y Relación con Software}

\textbf{Resumen e Introducción:} En la gestión académica universitaria, las asignaturas deben distribuirse entre un número finito de franjas horarias y recursos físicos limitados. La Teoría de Grafos permite representar incompatibilidades binarias y estudiar su asignación libre de colisiones mediante la \textbf{coloración de vértices} (Bondy \& Murty, 2008; Diestel, 2017; West, 2001). En la literatura de Ingeniería de Software e Investigación de Operaciones sobre el \textit{University Course Timetabling Problem} (UCTP), este enfoque transforma reglas operativas dispersas en un modelo combinatorio verificable (Carter et al., 1996; Schaerf, 1999; Burke \& Petrovic, 2002; Lewis, 2021).

\textbf{Problema y Justificación:} En la Carrera de Ingeniería de Software (FIE--ESPOCH), cuya malla curricular abarca 8 niveles PAO con entre 5 y 6 asignaturas por nivel (aproximadamente 45 cursos por período académico que comparten una planta docente y laboratorios de cómputo limitados), dos asignaturas que comparten estudiantes de un mismo nivel de paralelo único (como PAO 3), el mismo docente titular o un mismo laboratorio exclusivo no pueden dictarse simultáneamente. Gestionar estas restricciones de forma manual provoca cruces de horario, reprogramaciones al inicio del semestre y subutilización de aulas (Lewis, 2008; Pinedo, 2016).
\textbf{Pregunta de investigación:} \textit{¿Cómo representar formalmente las incompatibilidades de recursos y participantes mediante un grafo simple no dirigido $G = (V, E)$ y aplicar algoritmos propios de coloración de vértices (Welsh-Powell y DSatur) para construir una asignación horaria libre de conflictos que minimice las franjas utilizadas?}

\textbf{Objetivo General:} Diseñar y desarrollar un prototipo de software que modele los conflictos de simultaneidad académica como un grafo simple no dirigido $G = (V, E)$ y aplique los algoritmos de coloración de Welsh-Powell y DSatur implementados desde cero en Python para generar horarios libres de colisiones.

\textbf{Objetivos Específicos:}
\begin{enumerate}
    \item Formalizar las restricciones de simultaneidad como vértices $V$ y aristas $E$ mediante lógica de predicados y relaciones binarias, construyendo la lista y la matriz de adyacencia $A_G$.
    \item Demostrar formalmente tres proposiciones del modelo: la bien-definición del grafo simple e independencia de cada clase $C_i$, la cota inferior por clique de nivel PAO $\chi(G) \ge \omega(G) \ge |V_{\text{pao}}(s)|$ y la cota superior voraz $\chi(G) \le \Delta(G) + 1$.
    \item Implementar y comparar desde cero en Python 3 los algoritmos de \textbf{Welsh-Powell} (Fase 3) y \textbf{DSatur} (Sprint 2), justificando sus estructuras de datos y complejidad $O(|V|^2)$.
    \item Verificar computacionalmente con pruebas automatizadas que cada clase $C_i$ sea independiente y que toda arista $\{u, v\} \in E$ cumpla $c(u) \neq c(v)$.
    \item Construir una interfaz web modular (Sprint 3) que importe restricciones CSV/JSON y visualice la matriz $A_G$, el grafo coloreado y el horario resultante.
\end{enumerate}

\begin{definicion}[Grafo Simple de Conflictos y Predicado Lógico $P(u, v)$]
Sea $V = \{v_1, \dots, v_n\}$ el conjunto finito de asignaturas, cada una caracterizada por la tupla $(id, doc, pao, aula, grp)$, asumiendo explícitamente un único paralelo por nivel curricular $pao(v)$ (acorde a la oferta de PAO 3 en la carrera) y reservando $grp(v)$ para subgrupos específicos o electivas compartidas. Definimos el \textbf{predicado lógico de conflicto} $P(u, v)$ sobre $V \times V$ como:
\begin{align*}
P(u, v) \iff (u \neq v) \land \Big[ &\big(doc(u) = doc(v)\big) \lor \big(pao(u) = pao(v)\big) \\
&\lor \big(aula(u) = aula(v)\big) \lor \big(grp(u) \cap grp(v) \neq \emptyset\big) \Big]
\end{align*}
La relación binaria $R = \{(u, v) \in V \times V \mid P(u, v)\}$ define el conjunto de aristas no dirigidas $E = \{\{u, v\} \subseteq V \mid P(u, v)\}$ del grafo simple $G = (V, E)$, donde la vecindad de $v$ es $N_G(v) = \{u \in V \mid \{u, v\} \in E\}$ y su grado es $\deg_G(v) = |N_G(v)| = \sum_{j=1}^n a_{ij}$.
\end{definicion}

\begin{definicion}[Conjunto Independiente, Clique y $k$-Coloración Propia]
En un grafo simple $G = (V, E)$: (1) $I \subseteq V$ es un \textbf{conjunto independiente} si $\forall u, v \in I, \{u, v\} \notin E$; (2) $Q \subseteq V$ es un \textbf{clique} si $\forall u \neq v \in Q, \{u, v\} \in E$, con número de clique $\omega(G) = \max\{|Q|\}$; y (3) una función $c: V \to \{1, \dots, k\}$ es una \textbf{$k$-coloración propia} si $\forall \{u, v\} \in E, c(u) \neq c(v)$. El \textbf{número cromático} $\chi(G)$ es el menor entero $k \ge 1$ para el cual existe una $k$-coloración propia (Jensen \& Toft, 1995; Lewis, 2021).
\end{definicion}
\end{singlespace}

\section{Proposiciones Matemáticas y Demostraciones}

\begin{singlespace}
\begin{proposicion}[Bien-definición del Grafo Simple de Horarios e Independencia de Franjas]
La relación de incompatibilidad $R = \{(u, v) \in V \times V \mid P(u, v)\}$ del modelo es irreflexiva y simétrica, por lo que $G = (V, E)$ es un grafo simple bien definido sin lazos; además, una función sobreyectiva $c: V \to \{1, \dots, k\}$ es una $k$-coloración propia si y solo si las clases de color $C_i = c^{-1}(\{i\})$ forman una partición de $V$ en conjuntos independientes libres de colisiones de docente, PAO, aula y grupo.
\end{proposicion}
\end{singlespace}

\begin{demostracion}[Demostración (Estrategia: Prueba directa de propiedades relacionales y reducción al absurdo)]
\textbf{Paso 1 (Irreflexividad y simetría de $R$):} Para todo $v \in V$, la proposición $(v \neq v)$ es falsa, luego $P(v, v)$ es falso y $(v, v) \notin R$ (irreflexividad, garantizando $a_{ii} = 0$). Si $(u, v) \in R$, por simetría de la igualdad ($=$), de la desigualdad ($\neq$) y conmutatividad de la intersección ($grp(u) \cap grp(v) = grp(v) \cap grp(u)$), se cumple $P(v, u)$, luego $(v, u) \in R$ (simetría, garantizando $a_{ij} = a_{ji}$).
\textbf{Paso 2 (Partición e independencia):} Por ser $c$ función sobreyectiva, las preimágenes $C_i$ son no vacías, disjuntas dos a dos y $\bigcup_{i=1}^k C_i = V$. Supongamos por reducción al absurdo que $c$ es propia pero algún $C_i$ contiene dos cursos $u, v \in C_i$ con conflicto $P(u, v)$. Entonces $\{u, v\} \in E$ y $c(u) = i = c(v)$, contradiciendo que $c(u) \neq c(v)$ en toda arista. Recíprocamente, si cada $C_i$ es independiente, ningún par $\{u, v\} \in E$ comparte clase, luego $c(u) \neq c(v)$.
\end{demostracion}

\begin{singlespace}
\begin{proposicion}[Cota Inferior por Clique de Nivel Curricular PAO]
Sea $V_{\text{pao}}(s) = \{v \in V \mid pao(v) = s\}$ el subconjunto de asignaturas de un mismo nivel curricular de paralelo único $s$. Entonces $V_{\text{pao}}(s)$ induce un subgrafo completo (clique) $K_{|V_{\text{pao}}(s)|}$ en $G$, y el número cromático satisface la cota inferior:
\[
\chi(G) \ge \omega(G) \ge \max_{s} |V_{\text{pao}}(s)|
\]
\end{proposicion}
\end{singlespace}

\begin{demostracion}[Demostración (Estrategia: Construcción directa de clique y Principio del Palomar)]
\textbf{Paso 1:} Sean $u, v \in V_{\text{pao}}(s)$ dos asignaturas distintas ($u \neq v$). Por definición del subconjunto, $pao(u) = s = pao(v)$, lo cual hace verdadera la disyunción en el predicado $P(u, v)$ (Definición 2.1). En consecuencia, $\{u, v\} \in E$ para todo par de cursos distintos de $V_{\text{pao}}(s)$, formando un clique de cardinalidad $|V_{\text{pao}}(s)| \le \omega(G)$.
\textbf{Paso 2:} Por el Principio del Palomar, si se intentara colorear un clique $Q$ de tamaño $\omega(G)$ con $k < \omega(G)$ colores, al menos dos vértices adyacentes $u, v \in Q$ recibirían el mismo color $c(u) = c(v)$, lo que es imposible. Por tanto, $\chi(G) \ge \omega(G) \ge \max_s |V_{\text{pao}}(s)|$.
\end{demostracion}

\begin{singlespace}
\begin{proposicion}[Corrección de \texttt{WelshPowellColoring.solve} y Cota $\chi(G) \le k \le \Delta(G) + 1$]
Para todo grafo simple finito $G = (V, E)$, el procedimiento voraz por clases implementado en \texttt{WelshPowellColoring.solve} (Algoritmo \ref{alg:welsh_dsatur}) termina produciendo una $k$-coloración propia válida que satisface $\chi(G) \le k \le \Delta(G) + 1$.
\end{proposicion}
\end{singlespace}

\begin{demostracion}[Demostración (Estrategia: Invariante de independencia por clase y conteo de vecinos en clases previas)]
\textbf{Paso 1 (Terminación e independencia de cada clase $C_j$):} En cada iteración del bucle \texttt{while}, el primer vértice disponible en $L$ se asigna a la nueva clase $C_j$, reduciendo $|U|$ en al menos $1$; al ser $V$ finito, el algoritmo termina en $k \le |V|$ iteraciones cubriendo todo $V$. Además, un vértice $v$ solo se incorpora a $C_j$ si se cumple la guarda $N_G(v) \cap C_j = \emptyset$ (\texttt{not has\_conflict\_in\_class}). Por consiguiente, ningún par de vértices en $C_j$ es adyacente, cada $C_j$ es un conjunto independiente y la función $c$ resultante es una $k$-coloración propia.
\textbf{Paso 2 (Demostración de la cota $k \le \Delta(G) + 1$):} Sea $v \in C_k$ un vértice que recibió el último color $k$. Como en las iteraciones previas $1, 2, \dots, k-1$ el vértice $v$ fue examinado y no pudo ingresar a ninguna de las clases $C_1, C_2, \dots, C_{k-1}$, la guarda $N_G(v) \cap C_i = \emptyset$ debió resultar falsa para todo $i \in \{1, \dots, k-1\}$. Luego, $v$ posee al menos un vecino $u_i \in N_G(v) \cap C_i$ en cada una de las $k - 1$ clases anteriores. Como las clases $C_1, \dots, C_{k-1}$ son mutuamente disjuntas, los vértices $u_1, \dots, u_{k-1}$ son distintos entre sí, de donde se sigue que $\deg_G(v) \ge k - 1$, y por tanto:
\[
k \le \deg_G(v) + 1 \le \Delta(G) + 1 \implies \chi(G) \le k \le \Delta(G) + 1
\]
(Welsh \& Powell, 1967; Diestel, 2017).
\end{demostracion}

\section{Diseño Algorítmico, Complejidad y Limitaciones}

El grafo $G = (V, E)$ se implementa mediante listas de adyacencia con conjuntos hash (\texttt{Dict[str, Set[str]]}, consulta de adyacencia en $O(1)$ promedio) y matriz $A_G$ (Algoritmo \ref{alg:welsh_dsatur}).

\begin{singlespace}
\begin{algorithm}[H]
\caption{Procedimientos Voraces de Coloración: Welsh-Powell y DSatur}
\label{alg:welsh_dsatur}
\scriptsize
\begin{algorithmic}[1]
\Require Grafo simple no dirigido $G = (V, E)$ con grados $\deg_G(v)$
\Ensure Asignación propia $c: V \to \{1, \dots, k\}$ y clases independientes $\{C_1, \dots, C_k\}$
\Function{WelshPowell}{$G$}
    \State Ordenar $V$ en secuencia $L = (v_{\pi(1)}, \dots, v_{\pi(n)})$ tal que $\deg_G(v_{\pi(1)}) \ge \dots \ge \deg_G(v_{\pi(n)})$
    \State $k \gets 0$; \quad $U \gets V$ \Comment{$U$: subconjunto de vértices aún no coloreados}
    \While{$U \neq \emptyset$}
        \State $k \gets k + 1$; \quad $C_k \gets \emptyset$
        \ForAll{$v \in L$ (en orden) tal que $v \in U$}
            \If{$N_G(v) \cap C_k = \emptyset$} \State $c[v] \gets k$; \quad $C_k \gets C_k \cup \{v\}$; \quad $U \gets U \setminus \{v\}$ \EndIf
        \EndFor
    \EndWhile
    \State \Return $(c, \{C_1, \dots, C_k\})$
\EndFunction
\Function{DSatur}{$G$}
    \State $U \gets V$; \quad $S_{\text{vec}}[v] \gets \emptyset\ \forall v \in V$ \Comment{$S_{\text{vec}}[v] = \{c(u) \mid u \in N_G(v) \setminus U\}$}
    \For{$\text{paso} \gets 1$ \textbf{to} $|V|$}
        \State Seleccionar $v^* \in U$ que maximice $(d_{\text{sat}}(v) = |S_{\text{vec}}[v]|,\ \deg_G(v))$
        \State $k^* \gets \min \{k \in \mathbb{Z}^+ \mid k \notin S_{\text{vec}}[v^*]\}$; \quad $c[v^*] \gets k^*$; \quad $U \gets U \setminus \{v^*\}$
        \ForAll{$u \in N_G(v^*) \cap U$} \State $S_{\text{vec}}[u] \gets S_{\text{vec}}[u] \cup \{k^*\}$ \EndFor
    \EndFor
    \State \Return $c$
\EndFunction
\end{algorithmic}
\end{algorithm}
\end{singlespace}

\textbf{Complejidad, Tiempos Medidos y Limitaciones:} Construir $G$ evalúa $\binom{|V|}{2}$ pares en $O(|V|^2)$. Welsh-Powell ordena $V$ en $O(|V| \log |V|)$ y recorre candidatos en $O(|V|^2)$; DSatur ejecuta $|V|$ selecciones en $O(|V|)$ y actualiza vecinos en $O(|E|)$, resultando ambos en \textbf{tiempo $O(|V|^2)$} y \textbf{espacio $O(|V| + |E|)$} (Cormen et al., 2022). Aunque determinar $\chi(G)$ en general es NP-Completo (Karp, 1972; Garey et al., 1976), las pruebas empíricas ejecutadas sobre nuestro prototipo registran un tiempo de construcción de $0{,}20\text{ ms}$ y de coloreado con Welsh-Powell de $0{,}10\text{ ms}$ para el dataset ESPOCH ($|V|=6, |E|=8$), y de $69{,}65\text{ ms}$ (construcción) y $8{,}67\text{ ms}$ (coloreado) para una instancia sintética de facultad completa ($|V|=200, |E|=4\,080$). Como limitaciones reconocidas: (1) la Definición 2.1 asume un único paralelo por nivel $pao(v)$ (para extenderla a niveles con múltiples paralelos como \texttt{SW-3A} y \texttt{SW-3B}, basta sustituir $pao(v)$ por el par $(pao(v), paralelo(v))$); y (2) cada curso se modela como bloque indivisible sin fraccionamiento en múltiples sesiones semanales ni validación de aforo.
